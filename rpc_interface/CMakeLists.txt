####################
#  CMake packages  #
####################

find_package(gRPC REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)


################################
#  gRPC / Protobuf generation  #
################################

set(PROTOS protos/sushi_rpc.proto)

# In yocto environments, yocto grpc does not include gRPCPluginTargets.cmake
# when find_package(gRPC REQUIRED) is called.
if(CMAKE_CROSSCOMPILING)
    find_program(Protobuf_PROTOC_EXECUTABLE REQUIRED NAMES protoc)
    if(NOT TARGET protobuf::protoc)
        add_executable(protobuf::protoc IMPORTED)
    endif()
    set_target_properties(protobuf::protoc PROPERTIES
        IMPORTED_LOCATION "${Protobuf_PROTOC_EXECUTABLE}")
    find_program(_gRPC_CPP_PLUGIN grpc_cpp_plugin)
else()
    set(_gRPC_CPP_PLUGIN \$<TARGET_FILE:gRPC::grpc_cpp_plugin>)
endif()


# Intermediate library target with the generated protobuf objects
add_library(protos-lib OBJECT)
target_sources(protos-lib PRIVATE ${PROTOS})
target_link_libraries(protos-lib PUBLIC gRPC::grpc++)
target_include_directories(protos-lib PUBLIC ${CMAKE_CURRENT_BINARY_DIR})
protobuf_generate(TARGET protos-lib LANGUAGE cpp)
protobuf_generate(
    TARGET protos-lib
    LANGUAGE grpc
    GENERATE_EXTENSIONS .grpc.pb.h .grpc.pb.cc
    PLUGIN
    "protoc-gen-grpc=${_gRPC_CPP_PLUGIN}"
)

# ZMQ stub — generated by proto/protoc-gen-zmq
# On Apple/Windows (non-cross-compiling), run the protoc plugin inside a venv so the
# required protobuf Python package can be used without installing it system-wide.
if((APPLE OR WIN32) AND NOT CMAKE_CROSSCOMPILING)
    set(VENV_PATH "${CMAKE_CURRENT_BINARY_DIR}/protocZMQ_venv")

    if(WIN32)
        set(VENV_PYTHON "${VENV_PATH}/Scripts/python.exe")
        set(VENV_PIP    "${VENV_PATH}/Scripts/pip.exe")
    else()
        set(VENV_PYTHON "${VENV_PATH}/bin/python3")
        set(VENV_PIP    "${VENV_PATH}/bin/pip")
    endif()

    if(NOT EXISTS "${VENV_PYTHON}")
        message(STATUS "Creating Python venv for protoc-gen-zmq at ${VENV_PATH}")
        execute_process(
            COMMAND ${Python3_EXECUTABLE} -m venv ${VENV_PATH}
            COMMAND_ERROR_IS_FATAL ANY
        )
        execute_process(
            COMMAND ${VENV_PIP} install protobuf
            COMMAND_ERROR_IS_FATAL ANY
        )
    endif()

    # Generate a launch script, as #! is not supported on Windows
    if(WIN32)
        set(ZMQ_PLUGIN_WRAPPER "${CMAKE_CURRENT_BINARY_DIR}/protoc-gen-zmq-wrapper.bat")
        file(WRITE "${ZMQ_PLUGIN_WRAPPER}"
             "@echo off\r\n\"${VENV_PYTHON}\" \"${CMAKE_CURRENT_SOURCE_DIR}/protoc-gen-zmq\" %*\r\n")
    else()
        set(ZMQ_PLUGIN_WRAPPER "${CMAKE_CURRENT_BINARY_DIR}/protoc-gen-zmq-wrapper.sh")
        file(WRITE "${ZMQ_PLUGIN_WRAPPER}"
             "#!/bin/sh\n\"${VENV_PYTHON}\" \"${CMAKE_CURRENT_SOURCE_DIR}/protoc-gen-zmq\" \"$@\"\n")
        execute_process(COMMAND chmod +x "${ZMQ_PLUGIN_WRAPPER}")
    endif()

    set(ZMQ_PLUGIN_EXECUTABLE "${ZMQ_PLUGIN_WRAPPER}")
else()
    set(ZMQ_PLUGIN_EXECUTABLE "${CMAKE_CURRENT_SOURCE_DIR}/protoc-gen-zmq")
endif()

protobuf_generate(TARGET protos-lib
        LANGUAGE zmq
        GENERATE_EXTENSIONS _zmq.h
        PLUGIN "protoc-gen-zmq=${ZMQ_PLUGIN_EXECUTABLE}"
)


######################
#  Library target    #
######################

set(SUSHI_GRPC_SOURCES src/grpc_server.cpp
                       src/grpc_control_service.cpp
                       src/grpc_async_service_call_data.cpp )

set(SUSHI_ZMQ_SOURCES  src/zmq_control_service.cpp
                       src/zmq_server.cpp )

set(SUSHI_RPC_INCLUDE_DIRS ${CMAKE_CURRENT_BINARY_DIR}/protos
                           ${CMAKE_CURRENT_SOURCE_DIR}
                           ${PROJECT_SOURCE_DIR}/include
                           ${PROJECT_SOURCE_DIR}/src )

add_library(sushi_rpc STATIC ${SUSHI_GRPC_SOURCES} ${SUSHI_ZMQ_SOURCES})

target_include_directories(sushi_rpc PUBLIC include)
target_include_directories(sushi_rpc PRIVATE ${SUSHI_RPC_INCLUDE_DIRS})
target_link_libraries(sushi_rpc PUBLIC gRPC::grpc++ protos-lib elk::warningSuppressor elklog cppzmq)
target_compile_features(sushi_rpc PRIVATE cxx_std_20)

if(MSVC)    
    set(SUSHI_RPC_COMPILE_OPTIONS /W4)
else()
    set(SUSHI_RPC_COMPILE_OPTIONS -Wall -Wextra -fPIC)
endif()

target_compile_options(sushi_rpc PRIVATE ${SUSHI_RPC_COMPILE_OPTIONS})

##################
#  Install step  #
##################

install(FILES ${PROTOS} DESTINATION ${CMAKE_INSTALL_PREFIX}/share/sushi)
